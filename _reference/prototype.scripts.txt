/* ---------- 节点角色：只看 class_type，不看字段名 ---------- */
// 提示词类：用户在创作台打的字进这里（或者档位自己写），不该出现在"可勾节点"里
const PROMPT_CLASSES = ['CLIPTextEncode','CLIPTextEncodeSDXL','MusicGenTextEncode','T5TextEncode','PromptBuilder'];
// 上传类：参考图/参考视频/参考音频，可能有多个，由服务端按用户上传动态添加
const UPLOAD_CLASSES = ['LoadImage','LoadImageMask','LoadAudio','LoadVideo','VHS_LoadVideo'];
// 产物类：SaveXxx / PreviewXxx
// （用 out:true 标记，这里只是给文档/提示用）
// 判断一个节点是不是"能进档位的参数节点"
function nodeRole(n){
  if(!n) return 'param';
  if(n.out) return 'output';
  if(PROMPT_CLASSES.includes(n.cls)) return 'prompt';
  if(UPLOAD_CLASSES.includes(n.cls)) return 'upload';
  return 'param';
}
function nodeRoleLabel(n){
  const r = nodeRole(n);
  return r==='prompt' ? '提示词' : r==='upload' ? '上传参考' : r==='output' ? '出片' : '';
}

// 这个字段用什么控件？四类 + 特殊的比例
function controlOf(f){
  const t = fieldType(f);
  if(t==='bool')  return 'switch';
  if(t==='combo') return 'select';
  if(t==='multi') return 'multi';
  if(t==='int'||t==='float') return 'number';
  return fieldMeta(f).multi ? 'textarea' : 'text';
}

function isNumField(f){ const t=fieldType(f); return t==='int'||t==='float'; }
function coerceField(f, raw){
  if(raw===''||raw===undefined||raw===null) return '';
  const t = fieldType(f);
  if(t==='int')   { const n=Math.round(+raw); return isNaN(n)?'':n; }
  if(t==='float') { const n=+raw; return isNaN(n)?'':Math.round(n*1000)/1000; }
  return raw;
}

/* ---------- 从节点清单里推出"这个工作流能改哪些值" ---------- */
// 只把"参数节点"自己的格子摊平：提示词、上传、产物节点一律不算。
// 新工作流进来不需要改这里——它有什么就读到什么。
function paramNodes(wfName){
  const lib = WF_LIB[wfName];
  return lib ? lib.list.filter(n=>nodeRole(n)==='param').filter(n=>(n.w||[]).length) : [];
}
function wfFields(wfName){
  const out = [];
  paramNodes(wfName).forEach(n=>{
    (n.w||[]).forEach(f=>{
      if(out.some(x=>x.node===n.id && x.field===f)) return;
      out.push({ node:n.id, field:f, cls:n.cls });
    });
  });
  return out;
}
// 单个节点"能进档位的格子"
function nodeFields(nodeId, wfName){
  const lib = WF_LIB[wfName];
  if(!lib) return [];
  const n = lib.list.find(x=>x.id===String(nodeId));
  if(!n || nodeRole(n)!=='param') return [];
  return (n.w||[]).slice();
}

/* ---------- 谁出现在 ② 的清单里 ---------- */
// 只有"参数节点"能勾。提示词/上传/出片三类不出现在清单上——
// 不是因为不让改，而是因为改它们没意义：提示词由创作台填，参考图由用户上传（还可能好几个）。
function pickableNodes(wfName){
  return paramNodes(wfName);
}
function nodePickableFields(nodeId, wfName){ return nodeFields(nodeId, wfName); }
// 这个工作流里"被排除的节点"，用来在页面上说明清楚为什么没列出来
function excludedNodes(wfName){
  const lib = WF_LIB[wfName];
  if(!lib) return [];
  return lib.list.filter(n=>nodeRole(n)!=='param')
    .map(n=>({ id:n.id, cls:n.cls, role:nodeRole(n), label:nodeRoleLabel(n) }));
}
// 提示词格子（给"描述框绑到哪"用）——它不进档位设置，但需要一个绑定
const PROMPT_FIELDS = ['text','prompt','cells','genre'];
function isPromptLike(field){ return PROMPT_FIELDS.includes(field); }
// 这份配置里哪些节点被选进了参数区
function pickedNodeIds(wf){
  const ids = [];
  (wf && wf.fields || []).forEach(f=>{ if(f.node && !ids.includes(String(f.node))) ids.push(String(f.node)); });
  return ids;
}
// 某个节点里能被设置的那几格
function nodeSetFields(nodeId, wfName){
  const lib = WF_LIB[wfName];
  if(!lib) return [];
  const n = lib.list.find(x=>x.id===String(nodeId));
  if(!n) return [];
  return (n.w||[]).slice();
}

// 数据模型（一个工作流）：
//   original = 原始工作流的参数快照（只读，就是默认档）
//   fields   = 进档位设置的节点（从参数节点里勾）
//   grades   = 档位列表。第 1 档永远是 { locked:true }，代表"默认档 = 原始工作流"。

function wf(name, path, timeout, fields, original, grades, opt){
  return Object.assign({
    name, path, timeout,
    prompt:{ node:null, field:null },
    fields,
    original,
    grades,
    output:{ node:null, kind:'图片', multi:false },
    status:'ok',
  }, opt||{});
}

/* ---------- 能力状态判定 ---------- */
function capStatus(cap){
  if(!cap.workflows || !cap.workflows.length) return 'empty';
  if(cap.workflows.every(w=>w.status==='err')) return 'err';
  if(!cap.enabled) return 'off';
  if(cap.workflows.some(w=>w.status==='err')) return 'partial';
  return 'ok';
}
function capUsable(cap){ const s = capStatus(cap); return s==='ok' || s==='partial'; }

/* ---------- 按输入挑工作流（规则来自 config.yaml，页面不编辑） ---------- */
function pickWorkflow(cap, refs){
  const n = refs.length;
  for(const w of cap.workflows){
    if(w.status==='err') continue;
    const rule = (cap.rules||[]).find(r=>r.wf===w.name);
    if(!rule) continue;
    const t = rule.when;
    if(t==='默认') return { wf:w, why:'兜底默认' };
    const m = t.match(/(图片|音频|视频|总文件数量)\s*(=|≥|>|≤|<)\s*(\d+)/);
    if(m){
      const kind=m[1], op=m[2], v=+m[3];
      const cnt = kind==='图片'||kind==='总文件数量' ? n : kind==='音频' ? refs.filter(r=>r.kind==='audio').length : refs.filter(r=>r.kind==='video').length;
      const hit = op==='='?cnt===v:op==='≥'?cnt>=v:op==='>'?cnt>v:op==='≤'?cnt<=v:cnt<v;
      if(hit) return { wf:w, why:`命中「${t}」` };
    }
  }
  const fb = cap.fallback==='auto' ? cap.workflows.find(w=>w.status!=='err') : null;
  if(fb) return { wf:fb, why:'条件都不匹配，用兜底' };
  return { wf:null, why:'没有匹配的工作流' };
}

/* ---------- 档位 ---------- */
function capGrades(cap){
  if(!cap || !cap.workflows || !cap.workflows.length) return [];
  const ok = cap.workflows.find(x=>x.status!=='err' && x.grades && x.grades.length);
  if(ok) return ok.grades;
  const only = cap.workflows.find(x=>x.grades && x.grades.length);
  return only ? only.grades : [];
}
// 某一档最终要写进工作流的这组值：默认档返回原始值，其余返回它自己那份拷贝
function gradeParams(wf, g){
  if(!wf || !g) return {};
  if(g.locked) return JSON.parse(JSON.stringify(wf.original||{}));
  return Object.assign(JSON.parse(JSON.stringify(wf.original||{})), g.params||{});
}
// 这个档位字段是不是数值
function isNumGradeField(f){ return isNumField(f.field); }
// 档位摘要：宽高成对写成 1024×1024
function gradeDetail(g, fields, wf){
  if(!g) return '';
  const vals = wf ? gradeParams(wf, g) : Object.assign({}, g.params||{});
  const fs = (fields && fields.length) ? fields : Object.keys(vals).map(k=>({ field:k }));
  const used = [];
  const parts = [];
  fs.forEach(f=>{
    if(used.includes(f.field)) return;
    const v = vals[f.field];
    if(v===''||v===undefined||v===null) return;
    if(f.field==='width' && vals.height!==undefined){
      used.push('height'); parts.push(v+'×'+vals.height); return;
    }
    used.push(f.field);
    const lb = fieldLabel(f.field);
    parts.push(lb+' '+(Array.isArray(v)?v.join('/'):v));
  });
  return parts.join(' · ');
}

/* 老版本存下来的任务（没有 runs）转成新结构，免得升级后一片空白 */
function normalizeJobs(){
  let touched = false;
  (S.jobs||[]).forEach(j=>{
    if(Array.isArray(j.runs)){
      // 一批只认一个种子：老数据里每条不一样，统一成首发那个
      const sd = j.seed || (j.runs[0] && j.runs[0].seed) || Math.floor(Math.random()*1e9);
      if(j.seed!==sd || j.runs.some(r=>r.seed!==sd)){ j.seed = sd; j.runs.forEach(r=>{ r.seed = sd; }); touched = true; }
      return;
    }
    touched = true;
    const st = j.state || 'done';
    const files = (j.archive||[]).slice();
    const media = String(j.cap||'').startsWith('video') ? 'video' : String(j.cap||'').startsWith('audio') ? 'audio' : 'image';
    const stem = (j.prompt||'成品').replace(/[\\/:*?"<>|\s，。、；：！？]/g,'_').slice(0,10);
    const n = Math.max(1, files.length || j.count || 1);
    const seed = j.seed || Math.floor(Math.random()*1e9);
    j.seed = seed;
    j.runs = Array.from({length:n}).map((_,i)=>({
      id:(j.id||uid('job'))+'_r'+(i+1), seq: ++_runSeq,
      mode: i===0 ? 'normal' : 'queue',
      wf: j.wf||'', grade: j.grade||'', seed, media,
      state: (st==='running'||st==='queued') ? 'canceled' : st,
      pct: st==='done' ? 100 : (j.pct||0),
      t0: j.t0, t1: j.t1, err: j.err,
      file: files[i] || (stem+'_'+String(i+1).padStart(4,'0')+'_.png'), stem
    }));
    delete j.state; delete j.pct; delete j.assets; delete j.count; delete j.archive; delete j.err;
    if(!j.refs) j.refs = [];
  });
  if(touched) saveState();
}

function doneRuns(j){ return jobRuns(j).filter(r=>r.state==='done'); }
function pendRuns(j){ return jobRuns(j).filter(r=>r.state==='running'||r.state==='queued'); }

/* 这一档要跑多少步（只用来显示进度文案；音乐/语音这类没步数的给 28） */
function stepTotal(job, run){
  const cap = S.capabilities.find(c=>c.id===job.cap);
  const wf = cap && (cap.workflows||[]).find(w=>w.name===run.wf);
  if(!wf) return 28;
  const st = wfFields(wf.name).find(f=>f.field==='steps');
  if(!st) return 28;
  const g = (wf.grades||[]).find(x=>x.name===run.grade);
  const v = g ? gradeParams(wf,g)[st.field] : null;
  return (+v) || 28;
}
/* 原型里的模拟耗时：步数越多跑得越久（跟真实显卡的绝对时间无关） */
function runDurMs(job, run){
  return Math.min(14000, 1200 + stepTotal(job, run) * 140);
}
function queueAhead(run){
  // 正在跑的那一个也算在前面，不然第二个排队会显示“前面还有 0 个”
  return allRuns().filter(x=>(x.r.state==='running'||x.r.state==='queued') && (x.r.seq||0) < (run.seq||0)).length;
}

function jobCard(j){
  const runs = jobRuns(j);
  const base = runs[0];
  if(!base) return '';
  const cap = S.capabilities.find(c=>c.id===j.cap);
  const promptBox = `<div class="msg-u" style="background:#f6f8fb;color:#45505f;box-shadow:none;max-width:none;margin-bottom:10px">${esc(j.prompt)}</div>`;

  /* 首发还没出片：生成中 / 等执行 */
  if(base.state==='queued' || base.state==='running'){
    const pct = Math.round(base.pct||0);
    const st = stepTotal(j, base);
    return `<div class="job" id="${j.id}">
      <div class="head"><span class="dot a pulse"></span><b>${base.state==='queued'?'等待执行':'生成中'}</b>
        <span class="wf">· ${esc(j.wf)}</span><span class="gradechip">${esc(base.grade||'默认')}</span>
        <span style="flex:1"></span><span class="tiny muted">${j.id}</span></div>
      <div class="body">
        ${promptBox}
        <div class="tiny muted" data-jobstep>${base.state==='queued'?'ComfyUI 马上开始':`正在采样 · ${Math.round(pct*st/100)}/${st} 步`}</div>
        <div class="progress"><i data-jobbar style="width:${pct}%"></i></div>
        <div class="spread tiny muted"><span>${base.state==='queued'?'等待显卡空闲':'预计还需 '+Math.max(2,Math.round((runDurMs(j,base)/1000)*(100-pct)/100))+' 秒'}</span><b style="color:#2f6bff" data-jobpct>${pct}%</b></div>
      </div>
      <div class="foot"><span class="grow">第一次生成只会出一张 · 跑完可以「加入队列生成」往后接着加</span>
        <button class="btn sm" data-cancel="${j.id}">取消</button></div>
    </div>`;
  }

  /* 首发就失败了 */
  if(base.state==='fail'){
    const e = base.err || {};
    return `<div class="job fail" id="${j.id}">
      <div class="head"><span class="dot r"></span><b>失败</b><span class="wf">· ${esc(j.wf)}</span>
        <span class="gradechip">${esc(base.grade||'默认')}</span>
        <span style="flex:1"></span><span class="tiny muted">${esc(e.retriable?'可以再试一次':'要先改一下再跑')}</span></div>
      <div class="body" style="padding-bottom:0">${promptBox}</div>
      <div class="errbox" id="err-${j.id}">
        <div>${esc(e.msg||'')}</div>
        <div class="raw">${esc(e.raw||'')}</div>
      </div>
      <div class="foot"><span class="grow"></span>
        <button class="btn sm" data-raw="${j.id}">看原始报错</button>
        ${e.retriable?'<button class="btn sm primary" data-retry="'+j.id+'">重试一次</button>':'<button class="btn sm" data-nav="admin">去检查配置</button>'}
      </div>
    </div>`;
  }

  if(base.state==='canceled'){
    return `<div class="job" id="${j.id}"><div class="head"><span class="dot"></span><b>已取消</b><span class="wf">· ${esc(j.wf)}</span></div>
      <div class="body">${promptBox}</div></div>`;
  }

  /* 首发已出片：成品横着往后排，队列里的还占一个位置 */
  const ds = doneRuns(j);
  const ps = pendRuns(j);
  const fs = runs.filter(r=>r.state==='fail');
  const n = ds.length + ps.length;
  const cols = n<=1 ? 1 : n===2 ? 2 : 3;
  const grades = [...new Set(runs.filter(r=>r.state!=='canceled').map(r=>r.grade||'默认'))];
  const running = ps.find(r=>r.state==='running');
  const lastT = Math.max(j.t0||0, ...runs.filter(r=>r.t1).map(r=>r.t1));

  return `<div class="job done" id="${j.id}">
    <div class="head"><span class="dot g"></span><b>已完成</b><span class="wf">· ${esc(j.wf)}</span>
      <span class="gradechip">${esc(grades.join(' / '))}</span>
      <span style="flex:1"></span>
      ${ps.length?`<span class="qchip"><span class="dot a pulse"></span>队列 ${ps.length} 个${running?` · 正在出成品 ${ds.length+1}`:''}</span>`:''}
      <span class="tiny muted">用时 ${Math.round((lastT-(j.t0||lastT))/1000)} 秒</span></div>
    <div class="body">
      ${promptBox}
      <div class="grid tiles" style="grid-template-columns:repeat(${cols},minmax(0,1fr));max-width:${cols*250+(cols-1)*9}px">
        ${ds.map((r,i)=>{
          const label = '成品 ' + (i+1);
          const sub = runSizeText(j, r);          // 写这一档的尺寸，比写种子有用
          const badge = (r.grade && r.grade!==base.grade) ? `<span class="badge">${esc(r.grade)}</span>` : '';
          return r.media==='video'
            ? `<div class="tile vid"><div style="width:100%;height:100%">${mockVideo(r.seed)}</div>${badge}
                <div class="acts"><button title="看大图" data-zoom="${j.id}|${i}">⤢</button><button title="下载" data-dl="${j.id}">⬇</button></div></div>`
            : `<div class="tile"><div style="position:absolute;inset:0">${mockArt(r.seed,label,sub)}</div>${badge}
                <div class="acts"><button title="看大图" data-zoom="${j.id}|${i}">⤢</button><button title="下载" data-dl="${j.id}">⬇</button>
                <button title="加入队列生成（同种子、更高档）" data-qadd="${j.id}">🕒</button><button title="复制种子" data-seed="${r.seed}">⧉</button></div></div>`;
        }).join('')}
        ${ps.map((r,k)=>`<div class="tile q" data-runprog="${r.id}">
          <button class="qx" title="撤掉这一档" data-qcancel="${r.id}">✕</button>
          <div class="qwrap">
            <div class="qlb">成品 ${ds.length+k+1}</div>
            <div class="progress"><i style="width:${Math.round(r.pct||0)}%"></i></div>
            <div class="tiny muted" data-runstep="${r.id}">${r.state==='running'?'正在采样…':'排队中 · 前面还有 '+queueAhead(r)+' 个'}</div>
            <div class="qpct" data-runpct="${r.id}">${Math.round(r.pct||0)}%</div>
          </div></div>`).join('')}
      </div>
      ${fs.length?`<div class="jobfail">${fs.map((r,i)=>`<div>有一条队列任务没跑成（${esc(r.grade||'默认')}）：${esc(r.err&&r.err.msg||'未知错误')}</div>`).join('')}</div>`:''}
    </div>
    <div class="foot">
      <span class="grow seedline"><i class="sdot"></i>种子 <b class="mono">${j.seed||base.seed}</b></span>
      <button class="btn sm qbtn" data-qadd="${j.id}" title="挑一档排到这批后面，再出一张">🕒 加入队列生成</button>
      <button class="btn sm" data-again="${j.id}">↻ 再来一张</button>
    </div>
  </div>`;
}

      return `<div class="composer"><div class="composer-inner">

  return `<div class="composer"><div class="composer-inner">
    <div class="parambar">
      ${requiresRef?`<button class="pb ${UI.refs.length?'on':''}" id="refBtn">🖼 参考图 ${UI.refs.length?`<span class="val">${UI.refs.length}</span>`:'<span class="muted tiny">必填</span>'}</button>`:''}

        <button class="sendbtn" id="sendBtn" title="发送">↑</button>

function needsRef(cap){
  return ['image.reference','image.lineart','image.sketch','sheet.character','video.reference','video.firstlast','audio.music.ref','audio.clone'].includes(cap.id);
}
function placeholderFor(cap){
  const m = {
    'image.generate':'描述你想要的画面，例如：雨夜霓虹街道，赛博朋克，湿地面反光',
    'video.firstlast':'描述这段过渡里发生什么，例如：镜头缓慢推近，光线逐渐变暗',
    'board.9':'一句话说清故事，例如：小机器人找猫，9 格分镜',
    'audio.tts':'把要读的文字粘进来',
    'audio.music':'描述想要什么音乐，例如：低沉的钢琴配雨声，慢速，适合当背景',
  };
  return m[cap.id] || '描述你想生成的内容…';
}
function pickHintFor(cap){
  if(cap.id==='image.generate') return UI.refs.length===0?'只有文字，直接画':'拿这张图来改';
  if(cap.id==='video.firstlast') return '需要首帧和尾帧两张图';
  return '';
}

/* ---------- 创作台交互 ---------- */
function bindStudio(){
  $$('[data-cap]').forEach(el=>el.onclick=()=>{
    UI.capId = el.dataset.cap; UI.params={}; UI.refs=[];
    const cap = currentCap();
    if(!capUsable(cap)||!cap.enabled) toast('这个能力还没配置，先去管理台配一下');
    render();
    window.scrollTo({top:0});
  });
  $$('[data-sample]').forEach(el=>el.onclick=()=>{
    const t = $('#promptInput'); if(t){ t.value = el.dataset.sample; t.focus(); }
  });
  const send = $('#sendBtn');
  if(send){ send.onclick = doSend; }
  const t = $('#promptInput');
  if(t){
    t.onkeydown = e=>{ if((e.metaKey||e.ctrlKey)&&e.key==='Enter'){ e.preventDefault(); doSend(); } };
    t.oninput = ()=>{ $('#inputbox').classList.remove('err'); };
  }
  const addRefBtn = $('#addRef');
  if(addRefBtn) addRefBtn.onclick = addRef;
  $$('[data-rmref]').forEach(el=>el.onclick=()=>{
    UI.refs = UI.refs.filter(r=>r.id!==el.dataset.rmref); refreshComposer();
  });
  $$('[data-grade]').forEach(el=>el.onclick=()=>{
    UI.params.grade = el.dataset.grade;
    const capNow = currentCap();
    const g = capGrades(capNow).find(x=>x.name===el.dataset.grade);
    refreshComposer();
    if(g){
      const wfNow = pickWorkflow(capNow, UI.refs).wf || (capNow.workflows||[])[0];
      const d = gradeDetail(g, wfNow.fields, wfNow);
      toast('已切到「'+g.name+'」'+(d?'：'+d:''));
    }
  });
  const refBtn = $('#refBtn');
  if(refBtn) refBtn.onclick = addRef;

  // 任务卡交互
  $$('[data-cancel]').forEach(el=>el.onclick=()=>{
    const j = S.jobs.find(x=>x.id===el.dataset.cancel);
    if(j){ cancelJob(j); toast('已取消，并通知 ComfyUI 停止'); }
  });
  $$('[data-raw]').forEach(el=>el.onclick=()=>{
    const box = $('#err-'+el.dataset.raw); if(!box) return;
    box.classList.toggle('open');
    el.textContent = box.classList.contains('open')?'收起详情':'展开详情';
  });
  // 重试一次 = 普通生成，同样要过"ComfyUI 有没有任务在跑"这一关
  $$('[data-retry]').forEach(el=>el.onclick=()=>{
    const j = S.jobs.find(x=>x.id===el.dataset.retry);
    if(!j) return;
    const r = jobRuns(j)[0];
    if(submitNormal(j.cap, j.prompt, r.grade, r.wf, j.refs||[])) toast('已重新提交');
  });
  // 再来一张 = 普通生成（新开一张卡），忙的时候只提示、不排队
  $$('[data-again]').forEach(el=>el.onclick=()=>{
    const j = S.jobs.find(x=>x.id===el.dataset.again);
    if(!j) return;
    const r = jobRuns(j)[0];
    submitNormal(j.cap, j.prompt, r.grade, r.wf, j.refs||[]);
  });
  // 加入队列生成 = 唯一允许进队列的入口
  $$('[data-qadd]').forEach(el=>el.onclick=()=>openQueueModal(el.dataset.qadd));
  // 排队中/正在跑的那一格可以单独撤掉
  $$('[data-qcancel]').forEach(el=>el.onclick=()=>{
    const run = allRuns().map(x=>x.r).find(r=>r.id===el.dataset.qcancel);
    if(!run) return;
    const job = S.jobs.find(j=>jobRuns(j).includes(run));
    run.state = 'canceled';
    if(EXEC.run===run){ clearInterval(EXEC.timer); EXEC.timer=null; EXEC.run=null; EXEC.job=null; }
    saveState();
    if(job) renderCard(job);
    renderDrawer();
    pump();
    toast('已撤掉这一档');
  });
  $$('[data-seed]').forEach(el=>el.onclick=()=>{
    const t2=$('#promptInput'); if(t2){ t2.value = (t2.value?t2.value+' ':'')+`seed ${el.dataset.seed}`; t2.focus(); }
    toast('种子已复制，粘到输入框即可复现'); });
  $$('[data-dl]').forEach(el=>el.onclick=()=>toast('演示版不做真下载'));
  $$('[data-zoom]').forEach(el=>el.onclick=()=>{
    const [jid,idx]=el.dataset.zoom.split('|');
    const j = S.jobs.find(x=>x.id===jid);
    const r = doneRuns(j)[+idx]; if(!r) return;
    const cap = S.capabilities.find(c=>c.id===j.cap);
    openModal(`<div class="m-head"><h3>成品预览</h3><button class="btn ghost sm" data-close>✕</button></div>
      <div class="m-body" style="padding:0"><div style="aspect-ratio:1/1;background:#eef2f8">${mockArt(r.seed,'成品')}</div></div>
      <div class="m-foot" style="flex-wrap:wrap">
        <span class="tiny muted mono" style="width:100%;margin-bottom:8px;word-break:break-all">${esc(archivePathFor(cap, r.file||''))}</span>
        <span class="tiny muted">种子 ${r.seed} · 档位 ${esc(r.grade||'默认')}</span><span style="flex:1"></span>
        <button class="btn sm" data-close>关闭</button><button class="btn sm primary" data-close>下载 ⬇</button></div>`, true);
  });
  $$('[data-jump]').forEach(el=>el.onclick=()=>{
    const node = document.getElementById(el.dataset.jump);
    if(node){ node.scrollIntoView({block:'center',behavior:'smooth'}); node.style.outline='2px solid #2f6bff'; node.style.outlineOffset='3px';
      setTimeout(()=>{node.style.outline='none';},1200); }
  });
  const hb = $('#healthBtn');
  if(hb) hb.onclick = ()=>{ UI.comfyOnline=!UI.comfyOnline; render(); toast('已恢复连接'); };
}

function addRef(){
  if(UI.refs.length>=3){ toast('这个能力最多 3 张参考图'); return; }
  UI.refs.push({ id:uid('ref'), kind:'image', seed: Math.floor(Math.random()*99999) });
  refreshComposer();
}

function refreshComposer(){
  const node = $('.composer'); if(!node) return;
  const focused = document.activeElement && document.activeElement.id==='promptInput';
  const val = focused ? document.activeElement.value : '';
  node.outerHTML = viewComposer();
  if(focused){ const t2 = $('#promptInput'); if(t2){ t2.value = val; t2.focus(); } }
  bindStudio();
}

function scrollChatBottom(){
  const list = S.jobs.filter(j=>j.cap===UI.capId);
  const last = list.length ? document.getElementById(list[list.length-1].id) : null;
  if(last){ last.scrollIntoView({block:'end'}); return; }
  window.scrollTo({top:0});
}

/* ================= ComfyUI 执行 · 排队规则 =================
   照真实 ComfyUI 的行为模拟：同一时刻只跑一个任务。
   · 普通生成（发送 / 再来一张 / 重试）：有任务在跑 → 只提示、不执行、也不进队列。
   · 队列生成（加入队列生成）：只在"没有普通任务没跑完"时才能加；加完能接着加，一个一个跑。
*/
const EXEC = { job:null, run:null, timer:null };
let _runSeq = 0;

function allRuns(){
  const out = [];
  (S.jobs||[]).forEach(j=>(j.runs||[]).forEach(r=>out.push({j,r})));
  return out;
}
function runningEntry(){ return allRuns().find(x=>x.r.state==='running') || null; }
function queuedEntries(){ return allRuns().filter(x=>x.r.state==='queued').sort((a,b)=>(a.r.seq||0)-(b.r.seq||0)); }
function pendingNormal(){ return allRuns().find(x=>x.r.mode!=='queue' && (x.r.state==='queued'||x.r.state==='running')) || null; }

function busyText(x){
  const cap = S.capabilities.find(c=>c.id===x.j.cap);
  const nm = (cap?cap.name:x.j.capName) + ' · ' + (x.r.grade||'默认');
  const tail = x.r.state==='running' ? (Math.round(x.r.pct||0)+'%') : '刚提交还没开始';
  return `「${nm}」（${tail}）`;
}
function toastBlocked(title, detail, ms=4400){
  $$('.tblock').forEach(e=>e.remove());
  const el = document.createElement('div');
  el.className = 'tblock';
  el.innerHTML = `<div class="tb-i">⏳</div><div class="tb-t"><b>${esc(title)}</b><span>${esc(detail)}</span></div>`;
  document.body.appendChild(el);
  setTimeout(()=>{ el.classList.add('out'); setTimeout(()=>el.remove(),260); }, ms);
}

function guardNormal(){
  if(!UI.comfyOnline){ toast('ComfyUI 没连上，先把它打开'); return false; }
  const busy = runningEntry();
  if(busy){
    toastBlocked('正在生成中', busyText(busy)+' 还没跑完。等这张出来再点；想要同一张的高清版，用下面的「加入队列生成」。');
    return false;
  }
  return true;
}
function guardQueue(){
  if(!UI.comfyOnline){ toast('ComfyUI 没连上，先把 ComfyUI 启动起来'); return false; }
  const p = pendingNormal();
  if(p){
    toastBlocked('先等这张出来', busyText(p)+' 还在跑。它出来以后再点「加入队列生成」，不然你现在等的这张会排在后面。');
    return false;
  }
  // 排队上限写在配置文件里（service.maxQueue），页面上不给改
  const cap = Math.max(1, +(S.service.maxQueue) || 20);
  const qn = queuedEntries().filter(x=>x.j && x.j.id !== UI.qJobId).length;
  if(qn >= cap){
    toastBlocked('排队已满', `已经排了 ${qn} 个（最多 ${cap} 个）。等前面出来几个再加。`);
    return false;
  }
  return true;
}

function newRun(job, mode, wfName, gradeName, media){
  // 一批只有一个种子：首发随机生成，之后所有队列任务都沿用它
  if(!job.seed) job.seed = Math.floor(Math.random()*1e9);
  return {
    id: uid('run'), seq: ++_runSeq, mode, wf:wfName, grade:gradeName||'',
    seed: job.seed, media,
    state:'queued', pct:0, t0:Date.now(),
    stem: (job.prompt||'成品').replace(/[\\/:*?"<>|\s，。、；：！？]/g,'_').slice(0,10)
  };
}

/* 普通生成：首发的那一张 */
function submitNormal(capId, prompt, gradeName, wfName, refs){
  if(!guardNormal()) return false;
  const cap = S.capabilities.find(c=>c.id===capId) || currentCap();
  if(!cap) return false;
  const wf = (cap.workflows||[]).find(w=>w.name===wfName) || { output:{kind:'图片'} };
  const okind = (wf.output && wf.output.kind) || '图片';
  const media = okind==='视频' ? 'video' : okind==='音频' ? 'audio' : 'image';
  const job = { id:uid('job'), cap:capId, capName:cap.name, wf:wfName, prompt, refs:refs||[], t0:Date.now(), runs:[] };
  job.runs.push(newRun(job, 'normal', wfName, gradeName, media));
  S.jobs.push(job);
  saveState();
  UI.autoScroll = true;
  pump();
  render();
  return true;
}

/* 队列生成：接到同一张卡的后面 */
function addQueueRun(job, gradeName){
  const base = jobRuns(job)[0];
  if(!base) return;
  // 同一个种子、换一档参数：出来的是同一张构图的高清版，不是另画一张
  const run = newRun(job, 'queue', base.wf, gradeName, base.media);
  job.runs.push(run);
  saveState();
  const ahead = queueAhead(run);
  const idx = doneRuns(job).length + pendRuns(job).length;
  renderCard(job);
  pump();
  renderDrawer();
  toast(ahead>0 ? `已加入队列：前面还有 ${ahead} 个，跑完接到成品 ${idx}` : `开跑了，出的片会接到成品 ${idx}`, 2600);
}

function pump(){
  if(EXEC.run) return;
  const next = queuedEntries()[0];
  if(!next) return;
  startRun(next.j, next.r);
}

function startRun(job, run){
  EXEC.job = job; EXEC.run = run;
  run.state = 'running'; run.pct = 1; run.t0 = Date.now();
  saveState();
  renderCard(job); renderDrawer();
  const dur = runDurMs(job, run);
  const t0 = Date.now();
  clearInterval(EXEC.timer);
  EXEC.timer = setInterval(()=>{
    if(run.state!=='running'){ clearInterval(EXEC.timer); EXEC.timer=null; return; }
    const p = Math.min(100, ((Date.now()-t0)/dur)*100);
    if(p>=100){ clearInterval(EXEC.timer); EXEC.timer=null; finishRun(job, run); return; }
    if(p > (run.pct||0)){ run.pct = p; paintRun(job, run); }
  }, 240);
}

function finishRun(job, run){
  run.state = 'done'; run.pct = 100; run.t1 = Date.now();
  const idx = doneRuns(job).length;   // 含自己
  const ext = run.media==='video' ? 'mp4' : run.media==='audio' ? 'wav' : 'png';
  run.file = run.stem + '_' + String(idx).padStart(4,'0') + '_.' + ext;
  if(EXEC.run===run){ EXEC.run = null; EXEC.job = null; }
  saveState();
  renderCard(job); renderDrawer();
  const cap = S.capabilities.find(c=>c.id===job.cap);
  if(run.mode==='queue'){
    toast(`队列里的「${run.grade||'默认'}」出来了，已接到成品 ${idx}`, 2800);
  }else{
    toast('生成完成，已存到「'+S.service.archiveDir+'/'+(cap?cap.name:'')+'/」', 2600);
  }
  pump();
}

function cancelJob(job){
  const r = jobRuns(job).find(x=>x.state==='running'||x.state==='queued');
  if(!r) return;
  r.state = 'canceled';
  if(EXEC.run===r){ clearInterval(EXEC.timer); EXEC.timer=null; EXEC.run=null; EXEC.job=null; }
  saveState(); render(); pump();
}

/* 图片/视频/音频在"同一批"这句话里说法不同，别对着音频说"同一张" */
function batchWording(media){
  if(media==='video') return { same:'同一条', better:'更清晰' };
  if(media==='audio') return { same:'同一段', better:'音质更好' };
  return { same:'同一张', better:'更精细' };
}

/* 成品角上写这一档实际的尺寸，比重复写种子有用（一批只有一个种子） */

  const wf = cap && (cap.workflows||[]).find(w=>w.name===run.wf);

/* ---- 档位表（核心页面） ---- */
// 这一页只干一件事：把一个工作流拆成"默认档 + 若干改过的拷贝档"。
//   左列 ① 工作流文件 ② 参数节点（勾哪些进设置） ③ 被排除的节点（说明为什么不让设）
//   ④ 新档位区：勾好的参数在这里填一组值 + 档位名 + 【添加】
//         点【添加】= 把原始工作流深拷贝一份，这批值写进拷贝，拷贝就是这一档要调的工作流
//   ⑤ 档位列表：一行一档，第一行永远是"默认档 = 原始工作流"（只读、删不掉）；
//      其余每行 ✎ 编辑 = 展开这一档那份拷贝的全部参数
// 控件形态跟着参数本身的类型走：数字→输入框，开关→开关，枚举→下拉，多选→多选。

/* 一格参数渲染成对应控件 */
function ctlHtml(field, value, attr, opt){
  opt = opt || {};
  const kind = controlOf(field);
  const meta = fieldMeta(field);
  const dis = opt.disabled ? 'disabled' : '';
  const ph = opt.placeholder || '';
  const a = attr ? `${attr}="${attr.split('@')[0].replace('@','|')}"` : '';
  const v = value===undefined || value===null ? '' : value;
  if(kind==='switch'){
    const on = v===true || v==='true' || v===1;
    return `<span class="sw ${on?'on':''} ${dis}" ${attr} data-switch="1" role="switch" aria-checked="${on}"><i></i><em>${on?'开':'关'}</em></span>`;
  }
  if(kind==='select'){
    const opts = fieldOpts(field);
    const cur = String(v);
    const list = opts.includes(cur) ? opts : (cur!=='' ? [cur].concat(opts) : opts);
    return `<select class="input" ${attr} ${dis}>${list.map(o=>`<option value="${esc(o)}" ${String(o)===cur?'selected':''}>${esc(o)}</option>`).join('')}</select>`;
  }
  if(kind==='multi'){
    const opts = fieldOpts(field);
    const cur = Array.isArray(v) ? v.map(String) : (v?String(v).split(',').map(x=>x.trim()).filter(Boolean):[]);
    const list = [...new Set([...opts, ...cur])];
    return `<span class="multi ${dis}" ${attr}>${list.map(o=>`<label class="mopt"><input type="checkbox" value="${esc(o)}" ${cur.includes(String(o))?'checked':''} ${dis}><span>${esc(o)}</span></label>`).join('')}</span>`;
  }
  if(kind==='textarea'){
    return `<textarea class="input" rows="2" ${attr} ${dis} placeholder="${esc(ph)}">${esc(v)}</textarea>`;
  }
  // number / text
  const numeric = kind==='number';
  const stepAttr = numeric ? `step="${meta.step||(meta.type==='int'?1:0.05)}"` : '';
  const minAttr = meta.min!==undefined ? `min="${meta.min}"` : '';
  const maxAttr = meta.max!==undefined ? `max="${meta.max}"` : '';
  const typeAttr = numeric ? 'type="number"' : 'type="text"';
  return `<input class="input ${numeric?'mono':''}" ${typeAttr} ${stepAttr} ${minAttr} ${maxAttr} value="${esc(v)}" ${attr} ${dis} placeholder="${esc(ph)}">`;
}

/* 一格参数：标签 + 控件 + 类型角标 */
function ctlRow(field, value, attr, opt){
  opt = opt || {};
  const kind = controlOf(field);
  const meta = fieldMeta(field);
  const wide = (kind==='textarea' || (kind==='text' && String(value??'').length>16)) ? 'wide' : '';
  return `<div class="pfld ${wide} ${opt.cls||''}">
    <div class="plb">
      <span class="pn">${esc(fieldLabel(field))}</span>
      <i class="pk">${esc(field)}</i>
      <span class="ptype t-${kind}">${esc(CONTROL_LABEL[kind]||kind)}</span>
      ${meta.hint?`<span class="phint">${esc(meta.hint)}</span>`:''}
    </div>
    <div class="pctl">${ctlHtml(field, value, attr, opt)}</div>
  </div>`;
}

/* 把一个节点的几格参数整块渲染出来 */
function nodeBlock(node, fieldList, getVal, attrOf, opt){
  opt = opt || {};
  const live = fieldList.filter(f=>nodeSetFields(node.id, opt.wfName).includes(f));
  return `<div class="node-edit ${opt.readonly?'ro':''}">
    <div class="ne-hd">
      <span class="cid">${node.id}</span>
      <b>${esc(node.cls)}</b>
      <span class="tiny muted">${live.length} 个参数</span>
      <span style="flex:1"></span>
      ${opt.readonly?'<span class="rochip">只读 · 没改过</span>':''}
      ${opt.hdExtra||''}
    </div>
    <div class="ne-bd">
      ${live.map(f=>ctlRow(f, getVal(f), attrOf(f), {disabled:opt.readonly, wfName:opt.wfName})).join('')}
    </div>
  </div>`;
}

/* 这个工作流里所有文字类格子（给"描述框绑到哪"用） */
// 提示词节点不进 ② 的可勾清单，但描述框必须能绑到它——所以单独扫一遍
function allPromptFields(wfName){
  const out = [];
  const lib = WF_LIB[wfName];
  if(!lib) return out;
  lib.list.forEach(n=>{
    if(nodeRole(n)==='output' || nodeRole(n)==='upload') return;
    (n.w||[]).forEach(f=>{ if(isPromptLike(f)) out.push({ node:n.id, field:f, cls:n.cls }); });
  });
  return out;
}

/* 读出这个工作流的原始参数（= 默认档的值，只读） */
function readOriginal(wfName){
  const lib = WF_LIB[wfName];
  const o = {};
  if(!lib) return o;
  lib.list.forEach(n=>{
    (n.w||[]).forEach(f=>{
      if(f==='upload'||f==='image'||f==='audio') return;
      if(o[f]!==undefined) return;
      const t = fieldType(f);
      if(t==='int')        o[f] = (f==='width'||f==='height') ? 1024 : f==='steps' ? 28 : f==='seed' ? 0 : 1;
      else if(t==='float') o[f] = 1.0;
      else if(t==='bool')  o[f] = false;
      else if(t==='multi') o[f] = fieldOpts(f).length ? [fieldOpts(f)[0]] : [];
      else if(t==='combo') o[f] = fieldOpts(f)[0]||'';
      else o[f] = '';
    });
  });
  return o;
}

    e.stopPropagation();
    const wf = currentCap().workflows[UI.editingWfIndex];
    const i = +el.dataset.gdel;
    if(!wf.grades[i] || wf.grades[i].locked) return;
    const nm = wf.grades[i].name;
    wf.grades.splice(i,1);
    UI.gOpen = Math.min(i, wf.grades.length-1);
    saveState(); render();
    toast(`已删掉档位「${nm}」`);
  });
  // ④ 的【添加】：把原始工作流深拷贝一份，草稿里的值写进拷贝，成为新的一档
  const ga = $('[data-gadd]');
  if(ga) ga.onclick=()=>{
    const wf = currentCap().workflows[UI.editingWfIndex];
    if(!wf) return;
    if(!pickedNodeIds(wf).length){ toast('先勾一个参数节点'); return; }
    const draft = UI.draft || { name:'', values:{} };
    const nm = String(draft.name||'').trim();
    if(!nm){ toast('先给这一档起个名字'); const box=$('#newGradeName'); if(box) box.focus(); return; }
    if((wf.grades||[]).some(g=>g.name===nm)){ toast('这个名字已经有一档了，换一个'); return; }
    const copy = cloneWfParams(wf);              // 原始工作流的完整拷贝
    copy.values = Object.assign(copy.values, JSON.parse(JSON.stringify(draft.values||{})));
    wf.grades = wf.grades || [];
    wf.grades.push({
      name: nm, note:'',
      params: JSON.parse(JSON.stringify(draft.values||{})),
      copyFile: `${(S.service.copyDir||'./workflows/ComfyBridge').replace(/\/+$/,'')}/${wf.name.replace(/\.json$/,'')}__${nm}.json`
    });
    UI.gOpen = wf.grades.length-1;
    UI.draft = { name:'', values:{} };
    saveState(); render();
    const changed = Object.keys(draft.values||{}).length;
    toast(`已添加档位「${nm}」，创作台里多了一个按钮`);
  };
  // 描述框绑定（页面上不再提供下拉，读工作流时自动认；这里保留兜底）
  const pn = $('[data-pnode]');
  if(pn) pn.onchange=()=>{
    const wf=currentCap().workflows[UI.editingWfIndex];
    const [node,field] = pn.value.split('|');
    wf.prompt = Object.assign({ label:'描述', required:true }, wf.prompt, { node, field });
    saveState();
  };

  const sv = $('[data-savewf]');
  if(sv) sv.onclick = ()=>{ saveState(); toast('已保存'); render(); };
  const ts = $('[data-savecfg]');
  if(ts) ts.onclick = ()=>{ saveState(); toast('已保存'); render(); };
  const tc = $('#testConn');
  if(tc) tc.onclick = ()=>{ tc.textContent='连接中…'; setTimeout(()=>{ tc.textContent='测试连接'; toast(UI.comfyOnline?'连接成功：队列空闲':'连接失败：目标计算机积极拒绝'); }, 700); };
  $$('[data-s]').forEach(el=>el.onchange=()=>{
    let v = el.value;
    if(el.dataset.s==='archiveDir'){
      v = String(v||'').trim() || 'output/ComfyBridgeOutput';
      if(v.includes('..')){ toast('路径里不能有 ..'); render(); return; }
      if(v.includes('models')){ toast('不能放在 models 目录里'); render(); return; }
    }
    S.service[el.dataset.s] = v; saveState();
    if(el.dataset.s==='archiveDir'){ toast('成果目录已改成：'+v); render(); }
  });
  $$('[data-sw]').forEach(el=>el.onclick=()=>{
    const k = el.dataset.sw; S.service[k] = !S.service[k];
    el.classList.toggle('on', S.service[k]); saveState();
    if(k==='bindLan' && S.service[k]) toast('已开启，记得设口令');
  });
  const rr2 = $('#rerun');
  if(rr2) rr2.onclick=()=>{ toast('正在自检…'); setTimeout(()=>render(), 500); };
  $$('[data-discard]').forEach(el=>el.onclick=()=>{
    openModal(`<div class="m-head"><h3>放弃这些修改？</h3><button class="btn ghost sm" data-close>✕</button></div>
      <div class="m-body tiny">改过但没保存的，都会恢复成上次保存的样子。</div>
      <div class="m-foot"><button class="btn sm" data-close>取消</button><button class="btn sm danger" data-do-reset>放弃</button></div>`);
  });
}

/* ================= 弹层工具 ================= */
function openModal(html, noMaskClose){
  closeModal();
  const mask = document.createElement('div');
  mask.className='modal-mask'; mask.id='mask';
  mask.innerHTML = `<div class="modal ${html.length>900?'wide':''}">${html}</div>`;
  document.body.appendChild(mask);
  $$('[data-close]', mask).forEach(el=>el.onclick=closeModal);
  mask.onclick = e=>{ if(e.target===mask && !noMaskClose) closeModal(); };
  $$('[data-confirm-del]', mask).forEach(el=>el.onclick=()=>{
    S.capabilities = S.capabilities.filter(c=>c.id!==el.dataset.confirmDel);
    saveState(); closeModal(); render(); toast('已删除');
  });
  $$('[data-do-reset]', mask).forEach(el=>el.onclick=()=>resetState());
  return mask;
}
function closeModal(){ const m=$('#mask'); if(m) m.remove(); }
function closeAll(){ closeModal(); }
